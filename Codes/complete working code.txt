// Sensor readings use the ESP32 12-bit ADC.

// Normalized difference:
// 100 * (sensorA - sensorB) / (sensorA + sensorB)

constexpr float START_ERROR = 7.0f;
constexpr float STOP_ERROR  = 3.0f;

// Proportional response determines how quickly
// the desired target changes, not motor speed.

constexpr float KP = 0.18f;

// Limit each target update to 3 degrees.

constexpr float MAX_TARGET_CHANGE = 7.0f;

// Actual commanded servo movement is limited
// to 1 degree per SERVO_INTERVAL.

// ==========================================================
// 8. SENSOR READING
// ==========================================================

float readAverageADC(int pin) {

  uint32_t total = 0;

  for (int i = 0; i < ADC_SAMPLES; i++) {
    total += analogRead(pin);
    delayMicroseconds(150);
  }

  return (float)total / ADC_SAMPLES;
}

// ==========================================================
// 9. EXPONENTIAL FILTER
// ==========================================================

float smoothADC(float previous, float current) {

  return previous +
         FILTER_ALPHA * (current - previous);
}

// ==========================================================
// 10. NORMALIZED TRACKING ERROR
// ==========================================================

float calculateError(float a, float b) {

  float sum = a + b;

  if (sum < 1.0f) {
    return 0.0f;
  }

  return 100.0f * (a - b) / sum;
}

// ==========================================================
// 11. DIRECTION DETECTION
// ==========================================================

int confirmDirection(
    float error,
    int &active,
    int &candidate,
    int &counter) {

  float magnitude = fabsf(error);

  if (magnitude <= STOP_ERROR) {

    active = 0;
    candidate = 0;
    counter = 0;

    return 0;
  }

  int desired = (error > 0) ? 1 : -1;

  if (active == desired) {
    return active;
  }

  // Require a larger error to START movement.

  if (magnitude < START_ERROR) {

    active = 0;
    candidate = 0;
    counter = 0;

    return 0;
  }

  if (candidate == desired) {
    counter++;
  } else {
    candidate = desired;
    counter = 1;
  }

  if (counter >= DIRECTION_CONFIRM) {

    active = desired;
    candidate = 0;
    counter = 0;
  } else {
    active = 0;
  }

  return active;
}

// ==========================================================
// 12. SENSOR UPDATE
// ==========================================================

void updateSensors() {

  float t = readAverageADC(LDR_TOP_PIN);
  float b = readAverageADC(LDR_BOTTOM_PIN);
  float l = readAverageADC(LDR_LEFT_PIN);
  float r = readAverageADC(LDR_RIGHT_PIN);

  if (!sensorReady) {

    ldrTop = t;
    ldrBottom = b;
    ldrLeft = l;
    ldrRight = r;

    sensorReady = true;

  } else {

    ldrTop = smoothADC(ldrTop, t);
    ldrBottom = smoothADC(ldrBottom, b);
    ldrLeft = smoothADC(ldrLeft, l);
    ldrRight = smoothADC(ldrRight, r);
  }

  float avgLight =
      (ldrTop + ldrBottom +
       ldrLeft + ldrRight) / 4.0f;

  bool tooDark = avgLight < MIN_AVG_LIGHT;

  bool saturated =
      ldrTop >= SATURATION_LEVEL &&
      ldrBottom >= SATURATION_LEVEL &&
      ldrLeft >= SATURATION_LEVEL &&
      ldrRight >= SATURATION_LEVEL;

  lightValid = !tooDark && !saturated;

  errorV = calculateError(
      ldrTop,
      ldrBottom
  );

  errorH = calculateError(
      ldrLeft,
      ldrRight
  );
}

// ==========================================================
// 13. TRACKING CONTROL
// ==========================================================

void updateTracking() {

  if (!lightValid) {

    activeH = 0;
    activeV = 0;

    candidateH = 0;
    candidateV = 0;

    countH = 0;
    countV = 0;

    // Hold present commanded position.
    targetH = posH;
    targetV = posV;

    return;
  }

  int dirH = confirmDirection(
      errorH,
      activeH,
      candidateH,
      countH
  );

  int dirV = confirmDirection(
      errorV,
      activeV,
      candidateV,
      countV
  );

  // ---------- HORIZONTAL ----------

  if (dirH != 0) {

    float adjustment =
        KP * fabsf(errorH);

    adjustment = constrain(
        adjustment,
        0.0f,
        MAX_TARGET_CHANGE
    );

    targetH +=
        dirH * H_DIRECTION * adjustment;

  } else {

    // Prevent chasing tiny errors.
    targetH = posH;
  }

  // ---------- VERTICAL ----------

  if (dirV != 0) {

    float adjustment =
        KP * fabsf(errorV);

    adjustment = constrain(
        adjustment,
        0.0f,
        MAX_TARGET_CHANGE
    );

    targetV +=
        dirV * V_DIRECTION * adjustment;

  } else {

    targetV = posV;
  }

  // Enforce safe angle limits.

  targetH = constrain(
      targetH,
      (float)H_MIN,
      (float)H_MAX
  );

  targetV = constrain(
      targetV,
      (float)V_MIN,
      (float)V_MAX
  );
}

// ==========================================================
// 14. SMOOTH SERVO MOVEMENT
// ==========================================================

void updateServos() {

  int requestedH = (int)roundf(targetH);
  int requestedV = (int)roundf(targetV);

  // ---------- HORIZONTAL ----------

  if (posH < requestedH) {

    posH = min(
        posH + SERVO_STEP,
        requestedH
    );

    servoH.write(posH);

  } else if (posH > requestedH) {

    posH = max(
        posH - SERVO_STEP,
        requestedH
    );

    servoH.write(posH);
  }

  // ---------- VERTICAL ----------

  if (posV < requestedV) {

    posV = min(
        posV + SERVO_STEP,
        requestedV
    );

    servoV.write(posV);

  } else if (posV > requestedV) {

    posV = max(
        posV - SERVO_STEP,
        requestedV
    );

    servoV.write(posV);
  }
}

// ==========================================================
// DS18B20 TEMPERATURE (NON-BLOCKING)
// This function does not delay tracking/servo updates.
// ==========================================================

void updateTemperature(unsigned long now) {

  // Read only after the 12-bit conversion has completed.
  if (tempConversionPending &&
      now - lastTempRequest >= TEMP_CONVERSION_MS) {

    float reading = temperatureSensor.getTempCByIndex(0);

    if (reading != DEVICE_DISCONNECTED_C &&
        isfinite(reading) &&
        reading >= -55.0f && reading <= 125.0f) {
      tempC = reading;
      tempValid = true;
    } else {
      tempValid = false;
    }

    tempConversionPending = false;
  }

  // Start the next measurement without blocking the servos.
  if (!tempConversionPending &&
      now - lastTempRequest >= TEMP_INTERVAL) {
    temperatureSensor.requestTemperatures();
    lastTempRequest = now;
    tempConversionPending = true;
  }
}

// ==========================================================
// INA219 + 16x2 I2C LCD ADDITIONS
// Original LDR, servo and DS18B20 functions above unchanged.
// ==========================================================

bool i2cResponds(uint8_t address) {
  Wire.beginTransmission(address);
  return Wire.endTransmission() == 0;
}

void initializeI2CDevices() {
  Wire.begin(I2C_SDA_PIN, I2C_SCL_PIN);
  Wire.setClock(100000);

  // Same address-scanning approach as the supplied test sketch.
  Serial.println("=== I2C Scanner (SDA=21, SCL=22) ===");
  int count = 0;
  bool lcd27Found = false;
  bool lcd3FFound = false;
  bool inaFound = false;

  for (uint8_t address = 1; address < 127; ++address) {
    if (i2cResponds(address)) {
      Serial.print("I2C device at 0x");
      if (address < 16) Serial.print('0');
      Serial.println(address, HEX);
      ++count;
      if (address == 0x27) lcd27Found = true;
      if (address == 0x3F) lcd3FFound = true;
      if (address == INA_I2C_ADDR) inaFound = true;
    }
  }

  if (count == 0) Serial.println("No I2C devices found; check wiring.");

  if (lcd27Found) lcd = &lcd27;
  else if (lcd3FFound) lcd = &lcd3F;

  if (lcd != nullptr) {
    lcd->init();
    lcd->backlight();
    lcd->setCursor(0, 0);
    lcd->print("SOLAR TRACKER   ");
    lcd->setCursor(0, 1);
    lcd->print("LCD INITIALIZED ");
    Serial.println("LCD ready.");
  } else {
    Serial.println("LCD not found at 0x27/0x3F; tracking continues.");
  }

  // Never read an absent INA219: avoids misleading 0V/0A data.
  if (inaFound && ina219.begin()) {
    inaReady = true;
    Serial.println("INA219 ready at 0x40.");
  } else {
    Serial.println("INA219 not found at 0x40; tracking continues.");
  }
}

void updateElectrical() {
  if (!inaReady || !i2cResponds(INA_I2C_ADDR)) {
    inaValid = false;
    return;
  }

  // V- (load side) voltage plus shunt drop = source-side voltage.
  float bus_V = ina219.getBusVoltage_V();
  float shunt_mV = ina219.getShuntVoltage_mV();
  float current_mA = ina219.getCurrent_mA();
  float power_mW = ina219.getPower_mW();

  if (isfinite(bus_V) && isfinite(shunt_mV) &&
      isfinite(current_mA) && isfinite(power_mW)) {
    solarVoltage_V = bus_V + shunt_mV / 1000.0f;
    solarCurrent_mA = current_mA;
    solarPower_mW = power_mW;  // Power delivered to load side.
    inaValid = true;
  } else {
    inaValid = false;
  }
}

// Pad to 16 characters and only update changed LCD sections.
// This minimizes I2C traffic and avoids lcd.clear() flicker.
void lcdWriteRow(uint8_t row, const char* message) {
  if (lcd == nullptr || row > 1) return;
  static char previous[2][17] = {{0}, {0}};
  char padded[17];
  uint8_t i = 0;
  while (i < 16 && message[i] != '\0') {
    padded[i] = message[i];
    ++i;
  }
  while (i < 16) padded[i++] = ' ';
  padded[16] = '\0';

  for (int index = 0; index < 16;) {
    if (previous[row][index] == padded[index]) {
      ++index;
      continue;
    }
    int start = index;
    char changed[17];
    int count = 0;
    while (index < 16 && previous[row][index] != padded[index]) {
      changed[count++] = padded[index];
      previous[row][index] = padded[index];
      ++index;
    }
    changed[count] = '\0';
    lcd->setCursor(start, row);
    lcd->print(changed);
  }
}

void updateLCD(unsigned long now) {
  if (lcd == nullptr) return;

  char line1[17];
  char line2[17];
  bool electricalPage = ((now / LCD_PAGE_DURATION) % 2UL) == 0;

  if (electricalPage) {
    if (inaValid) {
      // Example: "V:13.50 I:0.25A" and "P:3.38W T:29.5C"
      snprintf(line1, sizeof(line1), "V:%5.2f I:%4.2fA",
               solarVoltage_V, solarCurrent_mA / 1000.0f);
      if (tempValid) {
        snprintf(line2, sizeof(line2), "P:%.2fW T:%.1fC",
                 solarPower_mW / 1000.0f, tempC);
      } else {
        snprintf(line2, sizeof(line2), "P:%.2fW T:ERR",
                 solarPower_mW / 1000.0f);
      }
    } else {
      snprintf(line1, sizeof(line1), "INA219 NO DATA");
      if (tempValid) snprintf(line2, sizeof(line2), "TEMP:%.1fC", tempC);
      else snprintf(line2, sizeof(line2), "TEMP:NO SENSOR");
    }
  } else {
    snprintf(line1, sizeof(line1), "H:%3d V:%3d", posH, posV);
    if (!lightValid) snprintf(line2, sizeof(line2), "LIGHT HOLD");
    else if (activeH == 0 && activeV == 0)
      snprintf(line2, sizeof(line2), "STABLE");
    else snprintf(line2, sizeof(line2), "TRACKING");
  }

  lcdWriteRow(0, line1);
  lcdWriteRow(1, line2);
}

// ==========================================================
// LCD NETWORK STATUS / IP OVERLAY
// - While Wi-Fi connects: display connection status.
// - Once reachable: keep browser IP visible until the dashboard is opened.
// - After that: preserve the existing alternating LCD pages and
//   show the IP again for 5 seconds every 20 seconds.
// - If the router is unavailable, display offline status. No fallback hotspot.
// Uses lcdWriteRow() so there is no extra lcd.clear() flicker.
// ==========================================================
void updateLCDWithNetwork(unsigned long now) {
  if (lcd == nullptr) return;

  bool stationConnected = (WiFi.status() == WL_CONNECTED);
  if (!stationConnected) {
    if (now - wifiStartMs < WIFI_CONNECT_TIMEOUT_MS) {
      lcdWriteRow(0, "WiFi Connecting");
      lcdWriteRow(1, "Please wait...");
    } else {
      lcdWriteRow(0, "WiFi Not Ready");
      lcdWriteRow(1, "Check 2.4G WiFi");
    }
    return;
  }

  bool showIp = !dashboardOpened ||
      ((now % IP_REMINDER_PERIOD_MS) < IP_REMINDER_SHOW_MS);

  if (showIp) {
    String ip = WiFi.localIP().toString();
    lcdWriteRow(0, "Web Dashboard IP");
    lcdWriteRow(1, ip.c_str());
    return;
  }

  // Original solar electrical and tracking pages are unchanged.
  updateLCD(now);
}

// ==========================================================
// MANUAL BROWSER RELAY CONTROL — NO BATTERY PROTECTION
// ==========================================================
// Active-LOW relay: LOW=ON, HIGH=OFF. Defaults to OFF in setup.
// If your specific relay module uses an active-HIGH input,
// set RELAY_ACTIVE_LOW = false above.
void setRelay(bool on) {
  // Always command the pin, even if already in the requested state.
  digitalWrite(RELAY_PIN,
               on ? (RELAY_ACTIVE_LOW ? LOW : HIGH)
                  : (RELAY_ACTIVE_LOW ? HIGH : LOW));
  if (relayOn != on) {
    relayOn = on;
    Serial.printf("[RELAY] Manually switched %s from browser\n", on ? "ON" : "OFF");
  }
}

// LOCAL WEB SERVER / ROUTER-ONLY WI-FI (NO WEB LOGIN)
// ==========================================================
String jsonFloat(float value, int digits, bool valid=true) {
  return (valid && isfinite(value)) ? String(value, digits) : String("null");
}

void handleDashboard() {
  dashboardOpened = true; // IP was seen and dashboard successfully opened.
  server.sendHeader("Cache-Control", "no-store");
  server.send_P(200, "text/html; charset=utf-8", DASHBOARD_HTML);
}

void handleTelemetry() {
  String j;
  j.reserve(580);
  j += "{\"solar_v\":"; j += jsonFloat(solarVoltage_V, 2, inaValid);
  j += ",\"solar_a\":"; j += jsonFloat(solarCurrent_mA / 1000.0f, 3, inaValid);
  j += ",\"solar_w\":"; j += jsonFloat(solarPower_mW / 1000.0f, 2, inaValid);
  j += ",\"temp_c\":"; j += jsonFloat(tempC, 2, tempValid);
  j += ",\"tracking\":\"";
  j += !lightValid ? "LIGHT HOLD" : ((activeH == 0 && activeV == 0) ? "STABLE" : "TRACKING");
  j += "\",\"h\":"; j += posH;
  j += ",\"v\":"; j += posV;
  j += ",\"top\":"; j += String(ldrTop, 0);
  j += ",\"bottom\":"; j += String(ldrBottom, 0);
  j += ",\"left\":"; j += String(ldrLeft, 0);
  j += ",\"right\":"; j += String(ldrRight, 0);
  j += ",\"error_h\":"; j += String(errorH, 1);
  j += ",\"error_v\":"; j += String(errorV, 1);
  j += ",\"relay_on\":"; j += relayOn ? "true" : "false";
  j += "}";
  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "application/json", j);
}

void handleLoadControl() {
  String command = server.arg("mode");
  if (command == "on") {
    setRelay(true);
  } else if (command == "off") {
    setRelay(false);
  } else {
    server.send(400, "text/plain", "Invalid mode. Use on or off.");
    return;
  }
  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "text/plain", relayOn ? "Relay ON" : "Relay OFF");
}

void beginLocalServer() {
  server.on("/", HTTP_GET, handleDashboard);
  server.on("/api/data", HTTP_GET, handleTelemetry);
  server.on("/api/load", HTTP_POST, handleLoadControl);
  server.onNotFound([]() {
      server.send(404, "text/plain", "Not found");
  });
  server.begin();
  Serial.println("[WEB] Local dashboard ready (no separate web login; private LAN only).");
}

// Connect ONLY to the configured router; never start a public ESP32 hotspot.
void initializeWifi() {
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.setAutoReconnect(true);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD); // non-blocking; tracking continues
  wifiStartMs = millis();
  lastWifiRetry = wifiStartMs;
  Serial.println("[WiFi] Connecting to configured network (no AP fallback).");
  beginLocalServer();
}

void maintainWifi(unsigned long now) {
  if (now - lastWifiCheck < 3000) return;
  lastWifiCheck = now;
  bool connected = (WiFi.status() == WL_CONNECTED);
  if (connected && !stationLoggedIn) {
    stationLoggedIn = true;
    dashboardOpened = false;
    Serial.print("[WiFi] Open in browser: http://");
    Serial.println(WiFi.localIP());
  } else if (!connected && stationLoggedIn) {
    stationLoggedIn = false;
    dashboardOpened = false;
    Serial.println("[WiFi] Network lost; retrying. Tracking continues.");
  }
  if (!connected && now - lastWifiRetry >= WIFI_CONNECT_TIMEOUT_MS) {
    lastWifiRetry = now;
    WiFi.reconnect();
    Serial.println("[WiFi] Reconnecting to configured router...");
  }
}

// ==========================================================

void setup() {

  // Added output: relay defaults OFF at startup before attaching servos/Wi-Fi.
  digitalWrite(RELAY_PIN, RELAY_ACTIVE_LOW ? HIGH : LOW);
  pinMode(RELAY_PIN, OUTPUT);

  Serial.begin(115200);

  delay(500);

  Serial.println();
  Serial.println("==================================");
  Serial.println(" ESP32 DUAL AXIS SOLAR TRACKER V3 ");
  Serial.println("==================================");

  analogReadResolution(12);

  analogSetAttenuation(ADC_11db);

  pinMode(LDR_TOP_PIN, INPUT);
  pinMode(LDR_BOTTOM_PIN, INPUT);
  pinMode(LDR_LEFT_PIN, INPUT);
  pinMode(LDR_RIGHT_PIN, INPUT);

  // DS18B20: GPIO 4 data + 4.7k pull-up to 3.3V.
  // Async mode avoids the default 750ms blocking conversion.
  temperatureSensor.begin();
  temperatureSensor.setResolution(12);
  temperatureSensor.setWaitForConversion(false);

  Serial.print("DS18B20 devices detected: ");
  Serial.println(temperatureSensor.getDeviceCount());

  temperatureSensor.requestTemperatures();
  lastTempRequest = millis();
  tempConversionPending = true;

  // Added I2C initialization. Original servo setup below unchanged.
  initializeI2CDevices();

  // Initialize PWM.

  servoH.setPeriodHertz(50);
  servoV.setPeriodHertz(50);

  servoH.attach(
      SERVO_H_PIN,
      SERVO_MIN_US,
      SERVO_MAX_US
  );

  servoV.attach(
      SERVO_V_PIN,
      SERVO_MIN_US,
      SERVO_MAX_US
  );

  // Center at startup.

  posH = 90;
  posV = 90;

  targetH = posH;
  targetV = posV;

  servoH.write(posH);
  servoV.write(posV);

  delay(1200);

  Serial.println("Servos initialized.");
  Serial.println("Calibrate direction before loading.");
  Serial.println("Tracking system started.");

  // New subsystem initialization, original setup above unchanged.
  initializeWifi();
  updateLCDWithNetwork(millis()); // Show Wi-Fi status immediately.
}

// ==========================================================
// 17. MAIN LOOP
// ==========================================================

void loop() {

  unsigned long now = millis();

  // ---------- SENSOR TASK ----------

  if (now - lastSensor >= SENSOR_INTERVAL) {

    lastSensor = now;

    updateSensors();
  }

  if (!sensorReady) {
    return;
  }

  // ---------- TRACKING TASK ----------

  if (now - lastTrack >= TRACK_INTERVAL) {

    lastTrack = now;

    updateTracking();
  }

  // ---------- SERVO TASK ----------

  if (now - lastServo >= SERVO_INTERVAL) {

    lastServo = now;

    updateServos();
  }

  // ---------- NON-BLOCKING TEMPERATURE TASK ----------
  updateTemperature(now);

  // ---------- INA219 TASK (1-second interval) ----------
  if (now - lastINA >= INA_INTERVAL) {
    lastINA = now;
    updateElectrical();
  }

  // ---------- LCD TASK (1-second interval) ----------
  if (now - lastLCD >= LCD_INTERVAL) {
    lastLCD = now;
    updateLCDWithNetwork(now);
  }

  // ---------- MONITOR TASK ----------

  if (now - lastPrint >= PRINT_INTERVAL) {

    lastPrint = now;

    printStatus();
  }

  // Wi-Fi tasks: original tracking timing remains unchanged.
  maintainWifi(now);
  server.handleClient();
}
